   169	async def download_files_async(urls: List[str], session: ClientSession = None,
   170	                               headers: dict = None) -> List[bytes]:
   171	    """
   172	    Downloads multiple files to a list of ``bytes``.
   173	
   174	    :param urls: The URLs of the files to download.
   175	    :param session: *(optional)* A ``ClientSession``. If no session is passed, a new one will be created.
   176	    :param headers: *(optional)* Request headers to send.
   177	    :return: The downloaded files as ``bytes``.
   178	    """
   179	    close_session = session is None
   180	    session = session if session else ClientSession()
   181	
   182	    async def download_one(url: str, headers: dict = None) -> bytes:
   183	        for _ in range(4):
   184	            async with session.get(url, headers=headers) as response:
   185	                if response.status == 200:
   186	                    return await response.read()
   187	                if 400 <= response.status < 500:
   188	                    response.raise_for_status()
   189	        response.raise_for_status()
   190	
   191	    tasks = [download_one(url, headers) for url in urls]
   192	    data = await asyncio.gather(*tasks)
   193	
   194	    if close_session:
   195	        await session.close()
   196	
   197	    return data

   292	def stitch_cubemap_faces(faces: List[Image.Image], face_size: int,
   293	                         stitching_method: CubemapStitchingMethod) -> Union[Image.Image, List[Image.Image]]:
   294	    """
   295	    Stitches the six faces of a cubemap into one image.
   296	
   297	    :param faces: A list of faces in the order front, right, back, left, top, bottom.
   298	    :param face_size: The side length of one face of the cubemap in pixels.
   299	    :param stitching_method: The stitching method.
   300	    :return: A stitched image, or ``faces`` if the stitching method is ``NONE``.
   301	    """
   302	    if stitching_method == CubemapStitchingMethod.NONE:
   303	        return faces
   304	    elif stitching_method == CubemapStitchingMethod.NET:
   305	        pano_width = 4 * face_size
   306	        pano_height = 3 * face_size
   307	        image = Image.new('RGB', (pano_width, pano_height))
   308	        image.paste(im=faces[0], box=(1 * face_size, 1 * face_size))
   309	        image.paste(im=faces[1], box=(2 * face_size, 1 * face_size))
   310	        image.paste(im=faces[2], box=(3 * face_size, 1 * face_size))
   311	        image.paste(im=faces[3], box=(0,             1 * face_size))
   312	        image.paste(im=faces[4], box=(1 * face_size, 0))
   313	        image.paste(im=faces[5], box=(1 * face_size, 2 * face_size))
   314	        return image
   315	    elif stitching_method == CubemapStitchingMethod.ROW:
   316	        image = Image.new('RGB', (6 * face_size, face_size))
   317	        for i in range(0, 6):
   318	            image.paste(im=faces[i], box=(i * face_size, 0))
   319	        return image
   320	    else:
   321	        raise ValueError("Unsupported stitching method")
